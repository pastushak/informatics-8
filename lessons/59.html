# Проєкт «PyPizza» — замовлення піци у трьох вікнах
from tkinter import *
from tkinter import messagebox

BG = "#FFF3E0"
BASES = {"Тонка основа": 100, "Класична основа": 120, "Сирна основа": 150}
TOPPINGS = {"Сир": 20, "М'ясо": 30, "Овочі": 15}
DRINKS = {"Без напою": 0, "Кока-кола": 15, "Сік": 25, "Мінеральна вода": 20}


def open_pizza():
    """Вікно 2: конструктор піци."""
    pizza = Toplevel(start)
    pizza.title("Створи свою піцу")
    pizza.geometry("420x520")
    pizza.configure(bg=BG)

    base = StringVar(value="Класична основа")
    toppings = {name: BooleanVar() for name in TOPPINGS}
    drink = StringVar(value="Без напою")

    def price():
        total = BASES[base.get()]
        for name, var in toppings.items():
            if var.get():
                total += TOPPINGS[name]
        total += DRINKS[drink.get()]
        return total

    def update():
        price_label.config(text=f"Вартість: {price()} грн")

    Label(pizza, text="Основа:", font=("Calibri", 14, "bold"), bg=BG).pack(anchor="w", padx=20, pady=(10, 0))
    for name, cost in BASES.items():
        Radiobutton(pizza, text=f"{name} ({cost} грн)", variable=base, value=name, bg=BG,
                    font=("Calibri", 13), command=update).pack(anchor="w", padx=40)
    Label(pizza, text="Начинка:", font=("Calibri", 14, "bold"), bg=BG).pack(anchor="w", padx=20, pady=(10, 0))
    for name, cost in TOPPINGS.items():
        Checkbutton(pizza, text=f"{name} (+{cost} грн)", variable=toppings[name], bg=BG,
                    font=("Calibri", 13), command=update).pack(anchor="w", padx=40)
    Label(pizza, text="Напій:", font=("Calibri", 14, "bold"), bg=BG).pack(anchor="w", padx=20, pady=(10, 0))
    for name, cost in DRINKS.items():
        Radiobutton(pizza, text=f"{name} ({cost} грн)", variable=drink, value=name, bg=BG,
                    font=("Calibri", 13), command=update).pack(anchor="w", padx=40)

    price_label = Label(pizza, text="", font=("Calibri", 16, "bold"), bg=BG, fg="#C0392B")
    price_label.pack(pady=10)
    Button(pizza, text="Оформити замовлення", font=("Calibri", 13), bg="#C0392B", fg="white",
           command=lambda: open_order(price())).pack()
    update()


def open_order(total):
    """Вікно 3: контактні дані."""
    order = Toplevel(start)
    order.title("Оформлення замовлення")
    order.geometry("360x300")
    order.configure(bg=BG)
    Label(order, text=f"До сплати: {total} грн", font=("Calibri", 16, "bold"), bg=BG).pack(pady=10)
    fields = {}
    for name in ("Ім'я", "Телефон", "Адреса"):
        Label(order, text=name + ":", bg=BG, font=("Calibri", 12)).pack()
        fields[name] = Entry(order, font=("Calibri", 12), width=30)
        fields[name].pack()

    def confirm():
        if any(e.get().strip() == "" for e in fields.values()):
            messagebox.showwarning("Увага", "Заповніть усі поля", parent=order)
            return
        name = fields["Ім'я"].get()
        messagebox.showinfo("Дякуємо!", f"{name}, замовлення прийнято!\nОчікуйте дзвінка.", parent=order)
        order.destroy()

    Button(order, text="Підтвердити", font=("Calibri", 13), bg="#27AE60", fg="white", command=confirm).pack(pady=15)


start = Tk()
start.title("PyPizza")
start.geometry("400x260")
start.configure(bg=BG)
Label(start, text="🍕 PyPizza", font=("Calibri", 28, "bold"), bg=BG, fg="#C0392B").pack(pady=20)
Label(start, text="Що замовимо сьогодні?", font=("Calibri", 14), bg=BG).pack()
row = Frame(start, bg=BG)
row.pack(pady=20)
Button(row, text="Піца", width=10, font=("Calibri", 14), command=open_pizza).pack(side=LEFT, padx=10)
Button(row, text="Суші", width=10, font=("Calibri", 14),
       command=lambda: messagebox.showinfo("Суші", "Меню суші — ваше додаткове завдання!")).pack(side=LEFT, padx=10)
start.mainloop()

Робимо · Практична робота

Оформлення замовлення, тестування, впровадження

Мета: завершити застосунок PyPizza і протестувати весь сценарій замовлення.

  1. Вікно замовлення

    Функція open_order(total): сума, поля ім'я, телефон, адреса, кнопка «Підтвердити».

  2. Перевірка полів

    Порожні поля → попередження; усе заповнено → «Ім'я, замовлення прийнято!» і закриття вікна.

  3. Тестування

    Пройдіть 3 різні замовлення від старту до підтвердження. Перевірте суми вручну.

  4. Додатково
    • Меню суші у власному вікні.
    • Телефон — лише цифри (phone.isdigit()).
    • Кількість піц (Spinbox) з перерахунком ціни.
