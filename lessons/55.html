# Проєкт «Визначення індексу маси тіла (ІМТ)»
from tkinter import *
from tkinter import messagebox


def calculate():
    """Обчислює ІМТ і визначає категорію."""
    try:
        weight = float(entry_weight.get())
        height = float(entry_height.get()) / 100      # см → м
    except ValueError:
        messagebox.showerror("Помилка", "Введіть вагу і зріст числами")
        return
    if weight <= 0 or height <= 0:
        messagebox.showwarning("Увага", "Значення мають бути більші за нуль")
        return
    bmi = weight / height ** 2
    if bmi < 18.5:
        category, color = "дефіцит маси", "#5DADE2"
    elif bmi < 25:
        category, color = "норма", "#58D68D"
    elif bmi < 30:
        category, color = "надлишкова вага", "#F5B041"
    else:
        category, color = "ожиріння", "#EC7063"
    result.config(text=f"ІМТ = {round(bmi, 1)}\n{category}", bg=color)
    # Зображення: images[category] = PhotoImage(file="norma.png") і т.д.
    # picture.config(image=images[category])


window = Tk()
window.title("Індекс маси тіла")
window.geometry("400x330")
window.resizable(False, False)

Label(window, text="Калькулятор ІМТ", font=("Arial", 18, "bold")).pack(pady=10)
Label(window, text="Вага, кг:", font=("Arial", 12)).pack()
entry_weight = Entry(window, font=("Arial", 12), justify="center")
entry_weight.pack()
Label(window, text="Зріст, см:", font=("Arial", 12)).pack()
entry_height = Entry(window, font=("Arial", 12), justify="center")
entry_height.pack()
Button(window, text="Визначити", font=("Arial", 12), command=calculate).pack(pady=10)
result = Label(window, text="", font=("Arial", 14), width=22, height=3)
result.pack()
picture = Label(window)
picture.pack()

window.mainloop()
ВажливоТаблиця категорій ІМТ розроблена для дорослих. У підлітків ІМТ оцінюють за віковими таблицями ВООЗ, і робить це лікар. Тестуйте програму на вигаданих даних дорослих людей — це навчальна задача з програмування.
