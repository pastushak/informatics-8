# Проєкт «Розв'язування квадратних рівнянь» ax² + bx + c = 0
from tkinter import *
from tkinter import messagebox
from math import sqrt


def solve():
    """Читає коефіцієнти, обчислює дискримінант і корені."""
    try:
        a = float(entry_a.get())
        b = float(entry_b.get())
        c = float(entry_c.get())
    except ValueError:
        messagebox.showerror("Помилка", "Введіть числа в усі три поля")
        return
    if a == 0:
        messagebox.showwarning("Увага", "a не може дорівнювати 0 — це не квадратне рівняння")
        return
    d = b ** 2 - 4 * a * c                     # дискримінант
    if d > 0:
        x1 = (-b + sqrt(d)) / (2 * a)
        x2 = (-b - sqrt(d)) / (2 * a)
        text = f"D = {d}\nДва корені: x₁ = {round(x1, 3)}, x₂ = {round(x2, 3)}"
    elif d == 0:
        x = -b / (2 * a)
        text = f"D = 0\nОдин корінь: x = {round(x, 3)}"
    else:
        text = f"D = {d} < 0\nДійсних коренів немає"
    result.config(text=text)


def clear():
    """Очищає поля для нового обчислення."""
    for e in (entry_a, entry_b, entry_c):
        e.delete(0, END)
    result.config(text="")


window = Tk()
window.title("Розв'язування квадратних рівнянь")
window.geometry("500x350")
window.resizable(False, False)
window.configure(bg="#EAF2F8")

Label(window, text="ax² + bx + c = 0", font=("Arial", 20, "bold"), bg="#EAF2F8").pack(pady=15)

frame = Frame(window, bg="#EAF2F8")
frame.pack()
entries = []
for i, name in enumerate(("a", "b", "c")):
    Label(frame, text=f"{name} =", font=("Arial", 14), bg="#EAF2F8").grid(row=0, column=i * 2, padx=5)
    e = Entry(frame, width=6, font=("Arial", 14))
    e.grid(row=0, column=i * 2 + 1, padx=5)
    entries.append(e)
entry_a, entry_b, entry_c = entries

buttons = Frame(window, bg="#EAF2F8")
buttons.pack(pady=15)
Button(buttons, text="Розрахувати", font=("Arial", 12), bg="#2E86C1", fg="white", command=solve).pack(side=LEFT, padx=5)
Button(buttons, text="Очистити", font=("Arial", 12), command=clear).pack(side=LEFT, padx=5)

result = Label(window, text="", font=("Arial", 14), bg="#EAF2F8")
result.pack(pady=10)

window.mainloop()
