# Проєкт «Розрахунок вартості розробки сайту»
from tkinter import *
from tkinter import ttk

PRICES = {"Односторінковий (лендінг)": 4000,
          "Багатосторінковий (корпоративний)": 8000,
          "Інтернет-магазин": 26000}


def calculate(*args):
    """Перераховує вартість за всіма обраними параметрами."""
    total = PRICES[site_type.get()]
    if design.get():
        total *= 1.5                     # дизайн: +50 %
    if mobile.get():
        total *= 1.1                     # адаптація: +10 %
    days = days_scale.get()
    if days < 10:
        total *= 1.2                     # термінове виконання (менше 10 днів): +20 %
    discount = int(discount_box.get().replace("%", ""))
    total *= (100 - discount) / 100
    result.config(text=f"Вартість: {round(total)} грн за {days} днів")


def hurry():
    """Кнопка «Хочу швидше» — мінімальний термін."""
    days_scale.set(7)
    calculate()


window = Tk()
window.title("Замовлення сайту")
window.geometry("700x560")
window.resizable(False, False)
window.configure(bg="#FFE5CC")
BG = "#FFE5CC"

Label(window, text="Вартість розробки сайту", font=("Arial", 20, "bold"), bg=BG).pack(pady=10)

Label(window, text="Який сайт вам потрібен?", font=("Arial", 13, "bold"), bg=BG).pack()
site_type = StringVar(value="Односторінковий (лендінг)")
for name in PRICES:
    Radiobutton(window, text=f"{name} — {PRICES[name]} грн", variable=site_type, value=name,
                bg=BG, font=("Arial", 12), command=calculate).pack(anchor="w", padx=200)

Label(window, text="Додаткові можливості:", font=("Arial", 13, "bold"), bg=BG).pack(pady=(10, 0))
design = BooleanVar()
mobile = BooleanVar()
Checkbutton(window, text="Розробити дизайн сайту (+50 %)", variable=design, bg=BG, font=("Arial", 12), command=calculate).pack(anchor="w", padx=200)
Checkbutton(window, text="Адаптація під смартфони (+10 %)", variable=mobile, bg=BG, font=("Arial", 12), command=calculate).pack(anchor="w", padx=200)

Label(window, text="Час на розробку, днів (менше 10 — +20 %):", font=("Arial", 13, "bold"), bg=BG).pack(pady=(10, 0))
days_scale = Scale(window, from_=7, to=14, orient=HORIZONTAL, length=300, bg=BG, command=calculate)
days_scale.set(14)
days_scale.pack()
Button(window, text="Хочу швидше", command=hurry).pack()

Label(window, text="Знижка:", font=("Arial", 13, "bold"), bg=BG).pack(pady=(10, 0))
discount_box = ttk.Combobox(window, values=["0%", "10%", "20%", "30%"], state="readonly", width=6)
discount_box.current(0)
discount_box.bind("<<ComboboxSelected>>", calculate)
discount_box.pack()

Button(window, text="Обчислити", font=("Arial", 12), bg="#E67E22", fg="white", command=calculate).pack(pady=10)
result = Label(window, text="", font=("Arial", 16, "bold"), bg=BG)
result.pack()

calculate()
window.mainloop()

Робимо · Практична робота

Розробка, тестування і впровадження калькулятора

Мета: реалізувати розрахунок вартості з усіма параметрами і перевірити його.

  1. Функція calculate()

    Базова ціна за типом → надбавки за послуги → надбавка за терміновість → знижка. Результат — у мітку.

  2. Миттєве оновлення

    Прив'яжіть calculate до всіх елементів вибору.

  3. «Хочу швидше»

    Кнопка ставить повзунок на 7 днів і перераховує.

  4. Тестування

    Ваші 3 ручні розрахунки з ДЗ — чи збігаються з програмою?

  5. Зображення додатково

    Картинка типу сайту змінюється разом з вибором.
